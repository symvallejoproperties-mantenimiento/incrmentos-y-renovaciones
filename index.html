${label}
${fmt(value)}
${sub}
`;document.getElementById('kpis').innerHTML=`
Incrementos
${card('Incrementos activos',incTotal)}${card('Incrementos aplicados',applied,'var(--green)')}${card('Incrementos no aplicados',notApplied,'var(--red)')}
Renovaciones
${card('Renovaciones activas',renTotal,'var(--orange)','Registros activos del año seleccionado')}
Seguimiento
${card('Firmadas',signed,'var(--green)','Sobre renovaciones activas')}${card('En proceso',renProcess,'var(--yellow)','Sobre renovaciones activas')}${card('Sin seguimiento',renPending,'var(--red)','Sobre renovaciones activas')}${card('No renovadas',renCancelled,'var(--red)','Histórico del periodo seleccionado')}
`;document.getElementById('count').textContent=`${fmt(a.length)} de ${fmt(state.all.length)} registros`; const incStates=group(incRows,r=>classStatus(r)),renStates=group(renActiveRows,r=>classStatus(r));monthlyComparison(incRows,'incremento','incMonth','incMonthTable');monthlyComparison(renActiveRows,'renovacion','renMonth','renMonthTable');donut('incStatus',incStates);chartTable('incStatusTable',incStates,'Estatus','Incrementos');donut('renStatus',renStates);chartTable('renStatusTable',renStates,'Estatus','Renovaciones activas');timelineByType(a,'incremento','incTimeline','incTimelineTable');timelineByType(a,'renovacion','renTimeline','renTimelineTable');quarterSummary('qInc',incRows,'incremento');quarterSummary('qRen',renActiveRows,'renovacion');alerts(a);gauges(pct(applied,incTotal),pct(signed,renTotal));renderTables()}
